# Bootwright owned routing: bootwright-lab-lab-libvirt-routing.service
[Unit]
Description=Bootwright routes and isolates the VLAN legs on owned trunks
Before=network-pre.target NetworkManager.service
Wants=network-pre.target

[Service]
Type=oneshot
RemainAfterExit=yes
ExecStart=/usr/sbin/nft -f /etc/bootwright/routing/bootwright-lab-lab-libvirt-routing.nft
ExecStartPost=/usr/sbin/nft list table inet bootwright-lab-lab-libvirt-routing
ExecStartPost=/usr/sbin/sysctl -w net.ipv4.ip_forward=1
ExecReload=/usr/sbin/nft -f /etc/bootwright/routing/bootwright-lab-lab-libvirt-routing.nft
ExecReload=/usr/sbin/nft list table inet bootwright-lab-lab-libvirt-routing
ExecReload=/usr/sbin/sysctl -w net.ipv4.ip_forward=1
ExecStopPost=-/usr/sbin/nft delete table inet bootwright-lab-lab-libvirt-routing

[Install]
WantedBy=multi-user.target
